let catlakCatalogBusy=false,catlakCatalogQueued=false,catlakCatalogLoadedAt=0;
function catlakCatalogApply(rows,role){
  rows=rows||[];
  if(rows[0]&&!rows[0].error)st.items=rows[0].data||st.items||[];
  if(rows[1]&&!rows[1].error)st.inv=rows[1].data||st.inv||[];
  if(rows[2]&&!rows[2].error)st.rolls=rows[2].data||st.rolls||[];
  if(rows[3]&&!rows[3].error)st.notes=rows[3].data||st.notes||[];
  if(rows[4]&&!rows[4].error)st.speciesRows=rows[4].data||st.speciesRows||[];
  if(rows[5]&&!rows[5].error)st.backgroundRows=rows[5].data||st.backgroundRows||[];
  if(rows[6]&&!rows[6].error)st.classRows=rows[6].data||st.classRows||[];
  if(rows[7]&&!rows[7].error)st.paths=rows[7].data||st.paths||[];
  if(role==='gm'&&rows[8]&&!rows[8].error)st.gmnotes=rows[8].data||st.gmnotes||[];
}
async function catlakLoadCatalogBackground(role,uid,force=false){
  if(!st.ses?.user||st.ses.user.id!==uid)return;
  if(catlakCatalogBusy){catlakCatalogQueued=true;return}
  if(!force&&Date.now()-catlakCatalogLoadedAt<2500)return;
  catlakCatalogBusy=true;
  try{
    let q=[
      S.from('catlak_items').select('*').order('created_at'),
      S.from('catlak_inventory').select('*').order('granted_at'),
      S.from('catlak_rolls').select('*').order('created_at',{ascending:false}).limit(150),
      S.from('catlak_species_powers').select('*').order('species_name').order('unlock_level'),
      S.from('catlak_species').select('*').order('sort_order'),
      S.from('catlak_backgrounds').select('*').order('sort_order'),
      S.from('catlak_classes').select('*').order('sort_order'),
      S.from('catlak_special_paths').select('*').order('sort_order')
    ];
    if(role==='gm')q.push(S.from('catlak_item_gm_notes').select('*'));
    const rows=await Promise.all(q);
    if(!st.ses?.user||st.ses.user.id!==uid)return;
    catlakCatalogApply(rows,role);catlakCatalogLoadedAt=Date.now();
    window.dispatchEvent(new CustomEvent('catlak:data-refreshed',{detail:{source:'catalog-background'}}));
    const baseTabs=new Set(['builder','rules','rolls','races','items']);
    const keep=typeof window.__catlakShouldPreserveCurrentView==='function'&&window.__catlakShouldPreserveCurrentView();
    if(baseTabs.has(st.tab)&&!keep)render();
  }catch(e){console.warn('CATLAK_CATALOG_REFRESH',e)}
  finally{
    catlakCatalogBusy=false;
    if(catlakCatalogQueued){
      catlakCatalogQueued=false;
      const nextUid=st.ses?.user?.id||'';
      setTimeout(()=>catlakLoadCatalogBackground(st.role,nextUid,true),180);
    }
  }
}
async function refresh(){
  if(!st.ses?.user)return;
  const u=st.ses.user.id;
  const rr=await S.from('catlak_roles').select('role').eq('user_id',u).maybeSingle();
  if(rr.error)throw rr.error;
  const role=rr.data?.role||'player';
  let cq=S.from('catlak_characters').select('*');
  if(role!=='gm')cq=cq.eq('owner_id',u);
  const cr=await cq.order('created_at');
  if(cr.error)throw cr.error;
  if(!st.ses?.user||st.ses.user.id!==u)return;
  st.role=role;st.chars=cr.data||[];
  if(st.role==='gm'&&['sheet','builder'].includes(st.tab))st.tab='gm';
  if(st.role!=='gm'&&guest()&&!st.chars.some(c=>c.owner_id===u))st.tab='builder';
  const keep=typeof window.__catlakShouldPreserveCurrentView==='function'&&window.__catlakShouldPreserveCurrentView();
  if(st.role!=='gm'&&st.tab==='sheet'&&app.querySelector('main')?.dataset.ccHardSheet==='1'){
    window.dispatchEvent(new CustomEvent('catlak:data-refreshed',{detail:{source:'core-fast'}}));
  }else if(keep){
    window.dispatchEvent(new CustomEvent('catlak:data-refreshed',{detail:{source:'core-fast'}}));
  }else render();
  catlakLoadCatalogBackground(role,u,false);
}
